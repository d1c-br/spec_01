Begin {C62A69F0-16DC-11CE-9E98-00AA00574A4F} frmBlindInt 
   Caption         =   "Заглушка межфланцевая"
   ClientHeight    =   3210
   ClientLeft      =   120
   ClientTop       =   465
   ClientWidth     =   10155
   OleObjectBlob   =   "frmBlindInt.frx":0000
   StartUpPosition =   1  'CenterOwner
End
Attribute VB_Name = "frmBlindInt"
Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Explicit
Private mTarget As Range, mLoading As Boolean

Public Sub ShowFor(ByVal targetCell As Range)
    Dim ws As Worksheet, r As Long, execStr As String
    On Error GoTo fail
    Set mTarget = targetCell
    Set ws = targetCell.Worksheet
    r = targetCell.row
    mLoading = True
    Me.Caption = TypeField("BLIND_INT", 2)
    Me.lblGost.Caption = TypeField("BLIND_INT", 4)
    FillFromTable Me.cboSteel, "tSteels", 1
    Me.cboTyp.Clear
    Me.cboTyp.AddItem "1"
    Me.cboTyp.AddItem "2"
    SelIf Me.cboTyp, CStr(ws.Cells(r, col("ТипФ")).Value)
    If Me.cboTyp.ListIndex < 0 Then Me.cboTyp.ListIndex = 0
    execStr = CStr(ws.Cells(r, col("Исполн")).Value)
    If InStr(execStr, "хвостовик") > 0 Then Me.optTail.Value = True Else Me.optTurn.Value = True
    SelIf Me.cboSteel, CStr(ws.Cells(r, col("Сталь")).Value)
    mLoading = False
    RefillDN
    SelIf Me.cboDN, NumTxt0(ws.Cells(r, col("DN")).Value)
    RefillPN
    SelIf Me.cboPN, NumTxt0(ws.Cells(r, col("PN")).Value)
    Preview
    Me.show vbModal
    Exit Sub
fail:
    MsgBox "Не удалось открыть форму: " & Err.Description, vbExclamation
End Sub

Private Function NumTxt0(ByVal v As Variant) As String
    If IsNumeric(v) And Not isEmpty(v) Then NumTxt0 = NumTxt(v)
End Function

Private Sub SelIf(ByVal cbo As Object, ByVal v As String)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If cbo.List(i) = v Then cbo.ListIndex = i: Exit Sub
    Next i
    cbo.ListIndex = -1
End Sub

Private Function CurTurn() As Boolean
    CurTurn = Me.optTurn.Value
End Function

Private Sub RefillDN()
    Dim cur As String
    cur = Me.cboDN.Value
    mLoading = True
    FillList Me.cboDN, BaseValues("BLIND_INT", 3, 1, CLng(Me.cboTyp.Value), 0, Empty)
    SelIf Me.cboDN, cur
    mLoading = False
End Sub

Private Sub RefillPN()
    Dim cur As String
    cur = Me.cboPN.Value
    mLoading = True
    If Me.cboDN.ListIndex >= 0 Then
        FillList Me.cboPN, BaseValues("BLIND_INT", 4, 1, CLng(Me.cboTyp.Value), 3, CDbl(Me.cboDN.Value))
    Else
        Me.cboPN.Clear
    End If
    SelIf Me.cboPN, cur
    mLoading = False
End Sub

Private Sub optTurn_Click()
    If Not mLoading Then Preview
End Sub

Private Sub optTail_Click()
    If Not mLoading Then Preview
End Sub

Private Sub cboTyp_Change()
    If Not mLoading Then RefillDN: RefillPN: Preview
End Sub

Private Sub cboDN_Change()
    If Not mLoading Then RefillPN: Preview
End Sub

Private Sub cboPN_Change()
    If Not mLoading Then Preview
End Sub

Private Sub cboSteel_Change()
    If Not mLoading Then Preview
End Sub

Private Sub Preview()
    If Me.cboTyp.ListIndex < 0 Or Me.cboDN.ListIndex < 0 Or Me.cboPN.ListIndex < 0 _
       Or Me.cboSteel.ListIndex < 0 Then
        Me.lblPreview.Caption = "— выберите параметры —"
    Else
        Me.lblPreview.Caption = BlindIntDes(CLng(Me.cboTyp.Value), CDbl(Me.cboDN.Value), _
            CDbl(Me.cboPN.Value), Me.cboSteel.Value, CurTurn)
    End If
End Sub

Private Sub btnOK_Click()
    If Me.cboTyp.ListIndex < 0 Or Me.cboDN.ListIndex < 0 Or Me.cboPN.ListIndex < 0 _
       Or Me.cboSteel.ListIndex < 0 Then
        MsgBox "Выберите тип, DN, PN и сталь.", vbExclamation
        Exit Sub
    End If
    WriteBlindInt mTarget, CLng(Me.cboTyp.Value), CDbl(Me.cboDN.Value), _
        CDbl(Me.cboPN.Value), Me.cboSteel.Value, CurTurn
    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub
