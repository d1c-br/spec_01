   Caption         =   "Заглушка эллиптическая"

   ClientLeft      =   120
   ClientTop       =   465
   ClientWidth     =   9360.001
   OleObjectBlob   =   "frmCap.frx":0000
   StartUpPosition =   1  'CenterOwner
End
Attribute VB_Name = "frmCap"
Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Explicit
Private mTarget As Range, mLoading As Boolean

Public Sub ShowFor(ByVal targetCell As Range)
    Dim ws As Worksheet, r As Long
    On Error GoTo fail
    Set mTarget = targetCell
    Set ws = targetCell.Worksheet
    r = targetCell.row
    mLoading = True
    Me.Caption = TypeField("CAP", 2)
    Me.lblGost.Caption = TypeField("CAP", 4)
    FillFromTable Me.cboSteel, "tSteels", 1
    FillList Me.cboD, BaseValues("CAP", 2, 0, Empty, 0, Empty)
    SelIf Me.cboD, NumTxt0(ws.Cells(r, col("D")).Value)
    mLoading = False
    RefillT
    SelIf Me.cboT, NumTxt0(ws.Cells(r, col("S")).Value)
    SelIf Me.cboSteel, CStr(ws.Cells(r, col("Сталь")).Value)
    If Len(Trim$(CStr(ws.Cells(r, col("Тип")).Value))) = 0 Then
        Me.chkPodk.Value = True
    Else
        Me.chkPodk.Value = (CStr(ws.Cells(r, col("Подк")).Value) = "П")
    End If
    Preview
    Me.show vbModal
    Exit Sub
fail:
    MsgBox "Не удалось открыть форму: " & Err.Description, vbExclamation
End Sub

Private Function NumTxt0(ByVal v As Variant) As String
    If IsNumeric(v) And Not isEmpty(v) Then NumTxt0 = NumTxt(v)
End Function

Private Sub SelIf(ByVal cbo As Object, ByVal v As String)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If cbo.List(i) = v Then cbo.ListIndex = i: Exit Sub
    Next i
    cbo.ListIndex = -1
End Sub

Private Sub RefillT()
    Dim cur As String
    cur = Me.cboT.Value
    mLoading = True
    If Me.cboD.ListIndex >= 0 Then
        FillList Me.cboT, BaseValues("CAP", 3, 2, CDbl(Me.cboD.Value), 0, Empty)
    Else
        FillList Me.cboT, BaseValues("CAP", 3, 0, Empty, 0, Empty)
    End If
    SelIf Me.cboT, cur
    mLoading = False
End Sub

Private Sub cboD_Change()
    If Not mLoading Then RefillT: Preview
End Sub

Private Sub cboT_Change()
    If Not mLoading Then Preview
End Sub

Private Sub cboSteel_Change()
    If Not mLoading Then Preview
End Sub

Private Sub chkPodk_Click()
    If Not mLoading Then Preview
End Sub

Private Sub Preview()
    If Me.cboD.ListIndex < 0 Or Me.cboT.ListIndex < 0 Or Me.cboSteel.ListIndex < 0 Then
        Me.lblPreview.Caption = "— выберите параметры —"
    Else
        Me.lblPreview.Caption = CapDes(CDbl(Me.cboD.Value), CDbl(Me.cboT.Value), _
            Me.cboSteel.Value, Me.chkPodk.Value)
    End If
End Sub

Private Sub btnOK_Click()
    If Me.cboD.ListIndex < 0 Or Me.cboT.ListIndex < 0 Or Me.cboSteel.ListIndex < 0 Then
        MsgBox "Выберите D, T и сталь.", vbExclamation
        Exit Sub
    End If
    WriteCap mTarget, CDbl(Me.cboD.Value), CDbl(Me.cboT.Value), Me.cboSteel.Value, Me.chkPodk.Value
    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub
