Attribute VB_Name = "frmElbow"
Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Explicit
Private mTarget As Range, mLoading As Boolean

Public Sub ShowFor(ByVal targetCell As Range)
    Dim ws As Worksheet, r As Long, pv As Variant
    On Error GoTo fail
    Set mTarget = targetCell
    Set ws = targetCell.Worksheet
    r = targetCell.row
    mLoading = True
    Me.Caption = TypeField("ELBOW", 2)
    Me.lblND.Caption = TypeField("ELBOW", 4)
    FillFromTable Me.cboSteel, "tSteels", 1
    FillList Me.cboD, BaseValues("ELBOW", 2, 0, Empty, 0, Empty)
    SelIf Me.cboD, NumTxt0(ws.Cells(r, col("D")).Value)
    mLoading = False
    RefillS
    SelIf Me.cboS, NumTxt0(ws.Cells(r, col("S")).Value)
    RefillAngle
    SelIf Me.cboAngle, NumTxt0(ws.Cells(r, col("Угол")).Value)
    SelIf Me.cboSteel, CStr(ws.Cells(r, col("Сталь")).Value)
    If Len(Trim$(CStr(ws.Cells(r, col("Тип")).Value))) = 0 Then
        Me.chkPodk.Value = True
    Else
        Me.chkPodk.Value = (CStr(ws.Cells(r, col("Подк")).Value) = "П")
    End If
    Preview
    Me.show vbModal
    Exit Sub
fail:
    MsgBox "Не удалось открыть форму: " & Err.Description, vbExclamation
End Sub
Private Function NumTxt0(ByVal v As Variant) As String
    If IsNumeric(v) And Not isEmpty(v) Then NumTxt0 = NumTxt(v)
End Function

Private Sub SelIf(ByVal cbo As Object, ByVal v As String)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If cbo.List(i) = v Then cbo.ListIndex = i: Exit Sub
    Next i
    cbo.ListIndex = -1
End Sub

Private Sub RefillS()
    Dim cur As String
    cur = Me.cboS.Value
    mLoading = True
    If Me.cboD.ListIndex >= 0 Then
        FillList Me.cboS, BaseValues("ELBOW", 3, 2, CDbl(Me.cboD.Value), 0, Empty)
    Else
        FillList Me.cboS, BaseValues("ELBOW", 3, 0, Empty, 0, Empty)
    End If
    SelIf Me.cboS, cur
    mLoading = False
End Sub

Private Sub RefillAngle()
    Dim cur As String, dOk As Boolean, sOk As Boolean
    cur = Me.cboAngle.Value
    dOk = (Me.cboD.ListIndex >= 0)
    sOk = (Me.cboS.ListIndex >= 0)
    mLoading = True
    If dOk And sOk Then
        FillList Me.cboAngle, BaseValues("ELBOW", 4, 2, CDbl(Me.cboD.Value), 3, CDbl(Me.cboS.Value))
    ElseIf dOk Then
        FillList Me.cboAngle, BaseValues("ELBOW", 4, 2, CDbl(Me.cboD.Value), 0, Empty)
    Else
        FillList Me.cboAngle, BaseValues("ELBOW", 4, 0, Empty, 0, Empty)
    End If
    SelIf Me.cboAngle, cur
    mLoading = False
End Sub

Private Sub cboD_Change()
    If mLoading Then Exit Sub
    RefillS
    RefillAngle
    Preview
End Sub

Private Sub cboS_Change()
    If mLoading Then Exit Sub
    RefillAngle
    Preview
End Sub

Private Sub cboAngle_Change()
    If mLoading Then Exit Sub
    Preview
End Sub

Private Sub cboSteel_Change()
    If mLoading Then Exit Sub
    Preview
End Sub

Private Sub chkPodk_Change()
    If mLoading Then Exit Sub
    Preview
End Sub

Private Sub Preview()
    If Me.cboD.ListIndex < 0 Or Me.cboS.ListIndex < 0 _
       Or Me.cboAngle.ListIndex < 0 Or Me.cboSteel.ListIndex < 0 Then
        Me.lblPreview.Caption = "— выберите все параметры —"
    Else
        Me.lblPreview.Caption = ElbowDes(CDbl(Me.cboD.Value), CDbl(Me.cboS.Value), _
                                         CDbl(Me.cboAngle.Value), Me.cboSteel.Value, _
                                         Me.chkPodk.Value)
    End If
End Sub

Private Sub btnOK_Click()
    If Me.cboD.ListIndex < 0 Or Me.cboS.ListIndex < 0 _
       Or Me.cboAngle.ListIndex < 0 Or Me.cboSteel.ListIndex < 0 Then
        MsgBox "Выберите D, S, угол и марку стали.", vbExclamation
        Exit Sub
    End If
    WriteElbow mTarget, CDbl(Me.cboD.Value), CDbl(Me.cboS.Value), _
               CDbl(Me.cboAngle.Value), Me.cboSteel.Value, Me.chkPodk.Value
    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub
