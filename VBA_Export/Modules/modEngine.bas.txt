Attribute VB_Name = "modEngine"
Option Explicit
Public mLastType As String

' ==================== Числа и токены ====================
Public Function NumTxt(ByVal v As Variant) As String
    NumTxt = CStr(CDbl(v))
End Function

Public Function NumKey(ByVal v As Variant) As String
    NumKey = Replace(NumTxt(v), ",", ".")
End Function

Public Function TokPodk(ByVal isPodk As Boolean) As String
    If isPodk Then TokPodk = "П " Else TokPodk = ""
End Function

Public Function TokSteelSeg(ByVal st As String) As String
    If Len(Trim$(st)) = 0 Or Trim$(st) = "20" Then
        TokSteelSeg = ""
    Else
        TokSteelSeg = "-" & st
    End If
End Function

' ==================== Обозначения ====================
Public Function TypeField(ByVal code As String, ByVal colNo As Long) As String
    TypeField = CStr(SprVal("tTypes", TypeRow(code), colNo))
End Function

Public Function BuildFromDict(ByVal code As String, ByVal tk As Object) As String
    Dim m As String, k As Variant
    m = TypeField(code, 3)
    For Each k In tk.keys
        m = Replace(m, "{" & CStr(k) & "}", CStr(tk(k)))
    Next k
    BuildFromDict = m
End Function

Public Function BuildDes(ByVal code As String, ByVal d As Variant, ByVal s As Variant, _
                         ByVal grp As String, ByVal st As String) As String
    Dim tk As Object
    Set tk = CreateObject("Scripting.Dictionary")
    tk("D") = NumTxt(d): tk("S") = NumTxt(s): tk("GRP") = grp: tk("ST") = st
    BuildDes = BuildFromDict(code, tk)
End Function

Public Function ElbowDes(ByVal d As Variant, ByVal s As Variant, ByVal ang As Variant, _
                         ByVal st As String, ByVal podk As Boolean) As String
    Dim tk As Object
    Set tk = CreateObject("Scripting.Dictionary")
    tk("D") = NumTxt(d): tk("S") = NumTxt(s): tk("ANG") = NumTxt(ang)
    tk("P") = TokPodk(podk)
    tk("STSEG") = TokSteelSeg(st)
    ElbowDes = BuildFromDict("ELBOW", tk)
End Function

' ==================== Выборки ====================
Public Function PipeValues(ByVal field As Long, ByVal fField As Long, ByVal fVal As Variant) As Collection
    Dim r As Long, c As New Collection, v As Variant, ok As Boolean
    UseBase "PIPE_SMLS"
    For r = 1 To BRows()
        ok = True
        If fField > 0 Then ok = (Abs(CDbl(BVal(r, fField)) - CDbl(fVal)) < 0.001)
        If ok Then
            v = BVal(r, field)
            On Error Resume Next
            c.Add v, CStr(v)
            On Error GoTo 0
        End If
    Next r
    Set PipeValues = c
End Function

Public Function BaseValues(ByVal code As String, ByVal field As Long, _
                           ByVal f1 As Long, ByVal v1 As Variant, _
                           ByVal f2 As Long, ByVal v2 As Variant) As Collection
    Dim r As Long, c As New Collection, v As Variant, ok As Boolean, cv As Variant
    UseBase code
    For r = 1 To BRows()
        ok = True
        If f1 > 0 Then
            cv = BVal(r, f1)
            If IsNumeric(v1) And IsNumeric(cv) Then
                ok = (Abs(CDbl(cv) - CDbl(v1)) < 0.001)
            Else
                ok = (CStr(cv) = CStr(v1))
            End If
        End If
        If ok And f2 > 0 Then
            cv = BVal(r, f2)
            If IsNumeric(v2) And IsNumeric(cv) Then
                ok = (Abs(CDbl(cv) - CDbl(v2)) < 0.001)
            Else
                ok = (CStr(cv) = CStr(v2))
            End If
        End If
        If ok Then
            v = BVal(r, field)
            On Error Resume Next
            c.Add v, CStr(v)
            On Error GoTo 0
        End If
    Next r
    Set BaseValues = c
End Function

Public Sub FillList(ByVal cbo As Object, ByVal c As Collection)
Dim v As Variant, arr() As Double, n As Long, i As Long, j As Long, tmp As Double
cbo.Clear
If c.Count = 0 Then Exit Sub
ReDim arr(1 To c.Count)
n = 0
For Each v In c
If IsNumeric(v) And Not isEmpty(v) Then
n = n + 1
arr(n) = CDbl(v)
End If
Next v
If n = 0 Then Exit Sub
For i = 1 To n - 1
For j = i + 1 To n
If arr(j) < arr(i) Then
tmp = arr(i): arr(i) = arr(j): arr(j) = tmp
End If
Next j
Next i
For i = 1 To n
cbo.AddItem NumTxt(arr(i))
Next i
End Sub

Public Sub FillFromTable(ByVal cbo As Object, ByVal tName As String, ByVal colNo As Long)
    Dim r As Long
    cbo.Clear
    For r = 1 To SprRows(tName)
        cbo.AddItem CStr(SprVal(tName, r, colNo))
    Next r
End Sub

Public Sub FillTypes(ByVal cbo As Object)
Dim r As Long, c As String, g As String, n As Long
Dim codes() As String, names() As String, ords() As Double, grps() As String
Dim i As Long, j As Long, ts As String, tD As Double
Dim curG As String
cbo.Clear
cbo.ColumnCount = 2
cbo.ColumnWidths = "0 pt;220 pt"
n = 0
ReDim codes(1 To SprRows("tTypes"))
ReDim names(1 To SprRows("tTypes"))
ReDim ords(1 To SprRows("tTypes"))
ReDim grps(1 To SprRows("tTypes"))
For r = 1 To SprRows("tTypes")
c = Trim$(CStr(SprVal("tTypes", r, 1)))
If Len(c) > 0 And Left$(c, 4) <> "SUP_" And Left$(c, 5) <> "PIPE_" Then
n = n + 1
codes(n) = c
names(n) = CStr(SprVal("tTypes", r, 2))
If IsNumeric(SprVal("tTypes", r, 10)) Then ords(n) = CDbl(SprVal("tTypes", r, 10)) Else ords(n) = 9999
g = Trim$(CStr(SprVal("tTypes", r, 11)))
If Len(g) = 0 Then g = "Прочее"
grps(n) = g
End If
Next r
For i = 1 To n - 1
For j = i + 1 To n
If ords(j) < ords(i) Then
tD = ords(i): ords(i) = ords(j): ords(j) = tD
ts = codes(i): codes(i) = codes(j): codes(j) = ts
ts = names(i): names(i) = names(j): names(j) = ts
ts = grps(i): grps(i) = grps(j): grps(j) = ts
End If
Next j
Next i
curG = ""
For i = 1 To n
If grps(i) <> curG Then
curG = grps(i)
cbo.AddItem ""
cbo.List(cbo.ListCount - 1, 1) = "— " & UCase$(curG) & " —"
End If
cbo.AddItem codes(i)
cbo.List(cbo.ListCount - 1, 1) = "    " & names(i)
Next i
End Sub

' ==================== Очистка и открытие форм ====================
Public Sub ClearPosParams(ByVal r As Long)
    Dim nm As Variant, i As Long, ws As Worksheet
    Set ws = ShSv()
        nm = Array("Тип", "D", "S", "Угол", "D2", "S2", "DN", "PN", "ТипФ", "Ряд", _
               "Исполн", "ГрК", "Группа", "Сталь", "Сталь2", "ТипО", "Высота", _
               "КлючБазы", "КлючПозиции", "Площадь", "Объём", "Длина", "ПарамАрм", _
               "DNсорт", "ПорядокГруппы", "Масса 1 ед", "Статус", "Подк")
    For i = 0 To UBound(nm)
        On Error Resume Next
        ws.Cells(r, col(CStr(nm(i)))).MergeArea.ClearContents
        On Error GoTo 0
    Next i
End Sub

Public Sub OpenByType(ByVal code As String, ByVal tgt As Range)
code = Trim$(code)
Select Case code
Case "VALVE": frmValve.ShowFor tgt
Case "GATE_VALVE_FL": frmValve.ShowFor tgt, "GATE_VALVE_FL"
Case "VALVE_M"
Dim fVM As Object
Set fVM = VBA.UserForms.Add("frmValveM")
fVM.ShowFor tgt
Unload fVM
Case "PIPE_SMLS": frmPipe.ShowFor tgt, "PIPE_SMLS"
Case "PIPE_SMLS_CD": frmPipe.ShowFor tgt, "PIPE_SMLS_CD"
Case "PIPE_WELD": frmPipe.ShowFor tgt, "PIPE_WELD"
Case "SUP_VP": frmSupportVP.ShowFor tgt
Case "SUP_TO": frmSupportTO.ShowFor tgt
Case "SUP_HB": frmSupportHB.ShowFor tgt
Case "SUP_KP": frmSupportKP.ShowFor tgt
Case "SUP_KH": frmSupportKH.ShowFor tgt
Case "ELBOW": frmElbow.ShowFor tgt
Case "TEE": frmTee.ShowFor tgt
Case "TRANS": frmTrans.ShowFor tgt
Case "FLANGE": frmFlange.ShowFor tgt
Case "CAP": frmCap.ShowFor tgt
Case "BLIND_FL": frmBlindFl.ShowFor tgt
Case "BLIND_INT": frmBlindInt.ShowFor tgt
Case Else: MsgBox "Неизвестный код [" & code & "]" & vbCrLf & CharCodes(code), vbExclamation
End Select
End Sub

Public Sub PickAndOpen(ByVal tgt As Range)
    Dim f As frmTypePick, code As String
    Set f = New frmTypePick
    f.Init mLastType
    f.show vbModal
    code = f.Picked
    Unload f
    If Len(code) = 0 Then Exit Sub
    mLastType = code
    OpenByType code, tgt
End Sub

' ==================== ТРУБЫ ====================
Public Function PipeRowEx(ByVal code As String, ByVal d As Double, ByVal s As Double) As Long
Dim r As Long
UseBase code
For r = 1 To BRows()
If IsNumeric(BVal(r, 1)) And IsNumeric(BVal(r, 2)) Then
If Abs(CDbl(BVal(r, 1)) - d) < 0.001 And Abs(CDbl(BVal(r, 2)) - s) < 0.001 Then
PipeRowEx = r: Exit Function
End If
End If
Next r
PipeRowEx = 0
End Function

Public Function PipeDes(ByVal code As String, ByVal d As Variant, ByVal s As Variant, _
ByVal grp As String, ByVal st As String) As String
Dim tk As Object, k As Long
Set tk = CreateObject("Scripting.Dictionary")
tk("D") = NumTxt(d)
tk("S") = NumTxt(s)
tk("s") = NumTxt(s)
tk("GRP") = grp
tk("ST") = st
If code = "PIPE_WELD" Then
k = PipeRowEx(code, CDbl(d), CDbl(s))
If k > 0 Then tk("TTG") = CStr(BVal(k, 7)) Else tk("TTG") = ""
End If
PipeDes = BuildFromDict(code, tk)
End Function

Public Sub WritePipeEx(ByVal tgt As Range, ByVal code As String, _
ByVal d As Double, ByVal s As Double, ByVal grp As String, ByVal st As String)
Dim ws As Worksheet, r As Long, k As Long, h As Long
Dim ev As Boolean, found As Boolean
ev = Application.EnableEvents
On Error GoTo fin
Set ws = ShSv()
r = tgt.row
h = HdrOf(r)
If h = 0 Then Err.Raise vbObjectError + 30, , "Строка " & r & " не входит в блок позиций."
UseBase code
Application.EnableEvents = False
ClearPosParams r
ws.Cells(r, col("Наименование")).Value = PipeDes(code, d, s, grp, st)
ws.Cells(r, col("Ед.изм.")).Value = TypeField(code, 6)
ws.Cells(r, col("Тип")).Value = code
ws.Cells(r, col("D")).Value = d
ws.Cells(r, col("S")).Value = s
ws.Cells(r, col("Группа")).Value = grp
ws.Cells(r, col("Сталь")).Value = st
ws.Cells(r, col("КлючБазы")).Value = code & "|" & NumKey(d) & "|" & NumKey(s)
ws.Cells(r, col("КлючПозиции")).Value = ws.Cells(r, col("КлючБазы")).Value & "|" & grp & "|" & st
ws.Cells(r, col("ПорядокГруппы")).Value = TypeField(code, 10)
k = PipeRowEx(code, d, s)
If k > 0 Then
ws.Cells(r, col("Масса 1 ед")).Value = BVal(k, 4)
ws.Cells(r, col("Масса 1 ед")).NumberFormat = "0.00"
ws.Cells(r, col("Площадь")).Value = BVal(k, 5)
ws.Cells(r, col("Объём")).Value = BVal(k, 6)
ws.Cells(r, col("DNсорт")).Value = BVal(k, 3)
found = True
End If
If found Then
ws.Cells(r, col("Статус")).Value = "ОК"
Else
ws.Cells(r, col("Масса 1 ед")).Value = "нет в базе"
ws.Cells(r, col("Статус")).Value = "НЕТ В БАЗЕ"
End If
If isEmpty(ws.Cells(r, col("№ввода")).Value) Then ws.Cells(r, col("№ввода")).Value = NextInputNo()
SetPosFormula r
ПеренумероватьБлок h
fin:
Application.EnableEvents = ev
If Err.Number <> 0 Then MsgBox "Ошибка записи: " & Err.Description, vbExclamation
End Sub

Public Sub WritePipe(ByVal tgt As Range, ByVal d As Double, ByVal s As Double, _
ByVal grp As String, ByVal st As String)
WritePipeEx tgt, "PIPE_SMLS", d, s, grp, st
End Sub

Public Sub WriteElbow(ByVal tgt As Range, ByVal d As Double, ByVal s As Double, _
                      ByVal ang As Double, ByVal st As String, ByVal podk As Boolean)
    Dim ws As Worksheet, r As Long, k As Long, h As Long
    Dim ev As Boolean, found As Boolean
    ev = Application.EnableEvents
    On Error GoTo fin
    Set ws = ShSv()
    r = tgt.row
    h = HdrOf(r)
    If h = 0 Then Err.Raise vbObjectError + 30, , "Строка " & r & " не входит в блок позиций."
    UseBase "ELBOW"
    Application.EnableEvents = False
    ClearPosParams r
    ws.Cells(r, col("Наименование")).Value = ElbowDes(d, s, ang, st, podk)
    ws.Cells(r, col("Ед.изм.")).Value = TypeField("ELBOW", 6)
    ws.Cells(r, col("Тип")).Value = "ELBOW"
    ws.Cells(r, col("D")).Value = d
    ws.Cells(r, col("S")).Value = s
    ws.Cells(r, col("Угол")).Value = ang
    ws.Cells(r, col("Сталь")).Value = st
    ws.Cells(r, col("Подк")).Value = IIf(podk, "П", "-")
    ws.Cells(r, col("КлючБазы")).Value = "ELBOW|" & NumKey(d) & "|" & NumKey(s) & "|" & NumKey(ang)
    ws.Cells(r, col("КлючПозиции")).Value = ws.Cells(r, col("КлючБазы")).Value & "|" & st & "|" & IIf(podk, "П", "-")
    ws.Cells(r, col("ПорядокГруппы")).Value = TypeField("ELBOW", 10)
    For k = 1 To BRows()
        If Abs(CDbl(BVal(k, 2)) - d) < 0.001 _
        And Abs(CDbl(BVal(k, 3)) - s) < 0.001 _
        And Abs(CDbl(BVal(k, 4)) - ang) < 0.001 Then
            ws.Cells(r, col("Масса 1 ед")).Value = BVal(k, 6)
            ws.Cells(r, col("Масса 1 ед")).NumberFormat = "0.00"
            ws.Cells(r, col("Площадь")).Value = BVal(k, 7)
            ws.Cells(r, col("Длина")).Value = BVal(k, 8)
            ws.Cells(r, col("DNсорт")).Value = BVal(k, 5)
            found = True
            Exit For
        End If
    Next k
    If found Then
        ws.Cells(r, col("Статус")).Value = "ОК"
    Else
        ws.Cells(r, col("Масса 1 ед")).Value = "нет в базе"
        ws.Cells(r, col("Статус")).Value = "НЕТ В БАЗЕ"
    End If
    If isEmpty(ws.Cells(r, col("№ввода")).Value) Then
        ws.Cells(r, col("№ввода")).Value = NextInputNo()
    End If
    SetPosFormula r
    ПеренумероватьБлок h
fin:
    Application.EnableEvents = ev
    If Err.Number <> 0 Then MsgBox "Ошибка записи: " & Err.Description, vbExclamation
End Sub

Public Function TeeRow(ByVal d As Double, ByVal t As Double, ByVal d1 As Double) As Long
    Dim r As Long
    UseBase "TEE"
    For r = 1 To BRows()
        If Abs(CDbl(BVal(r, 1)) - d) < 0.001 _
        And Abs(CDbl(BVal(r, 2)) - t) < 0.001 _
        And Abs(CDbl(BVal(r, 3)) - d1) < 0.001 Then
            TeeRow = r: Exit Function
        End If
    Next r
    TeeRow = 0
End Function

Public Function TeeVal(ByVal d As Double, ByVal t As Double, ByVal d1 As Double, ByVal colNo As Long) As Variant
    Dim k As Long
    k = TeeRow(d, t, d1)
    If k = 0 Then Err.Raise vbObjectError + 31, , "Нет такого тройника в базе tTees."
    UseBase "TEE"
    TeeVal = BVal(k, colNo)
End Function

Public Function TeeDes(ByVal d As Variant, ByVal t As Variant, ByVal d1 As Variant, _
                       ByVal T1 As Variant, ByVal st As String, ByVal podk As Boolean) As String
    Dim tk As Object
    Set tk = CreateObject("Scripting.Dictionary")
    tk("P") = TokPodk(podk)
    tk("D") = NumTxt(d)
    tk("T") = NumTxt(t)
    If Abs(CDbl(d1) - CDbl(d)) < 0.001 Then
        tk("BR") = ""
    Else
        tk("BR") = "-" & NumTxt(d1) & "х" & NumTxt(T1)
    End If
    tk("STSEG") = TokSteelSeg(st)
    TeeDes = BuildFromDict("TEE", tk)
End Function

' ==================== ПЕРЕХОДЫ ====================
Public Function TransValues(ByVal field As Long, ByVal d As Variant, ByVal t As Variant, _
                            ByVal d1 As Variant, ByVal T1 As Variant) As Collection
    Dim r As Long, c As New Collection, v As Variant, ok As Boolean
    UseBase "TRANS"
    For r = 1 To BRows()
        ok = True
        If Not isEmpty(d) Then ok = (Abs(CDbl(BVal(r, 1)) - CDbl(d)) < 0.001)
        If ok And Not isEmpty(t) Then ok = (Abs(CDbl(BVal(r, 2)) - CDbl(t)) < 0.001)
        If ok And Not isEmpty(d1) Then ok = (Abs(CDbl(BVal(r, 3)) - CDbl(d1)) < 0.001)
        If ok And Not isEmpty(T1) Then ok = (Abs(CDbl(BVal(r, 4)) - CDbl(T1)) < 0.001)
        If ok Then
            v = BVal(r, field)
            On Error Resume Next
            c.Add v, CStr(v)
            On Error GoTo 0
        End If
    Next r
    Set TransValues = c
End Function

Public Function TransRow(ByVal d As Double, ByVal t As Double, ByVal d1 As Double, _
                         ByVal T1 As Double, ByVal L As Double) As Long
    Dim r As Long
    UseBase "TRANS"
    For r = 1 To BRows()
        If Abs(CDbl(BVal(r, 1)) - d) < 0.001 _
        And Abs(CDbl(BVal(r, 2)) - t) < 0.001 _
        And Abs(CDbl(BVal(r, 3)) - d1) < 0.001 _
        And Abs(CDbl(BVal(r, 4)) - T1) < 0.001 _
        And Abs(CDbl(BVal(r, 6)) - L) < 0.001 Then
            TransRow = r: Exit Function
        End If
    Next r
    TransRow = 0
End Function

Public Function TransAutoT1(ByVal d As Double, ByVal t As Double, ByVal d1 As Double) As Double
    Dim r As Long
    UseBase "TRANS"
    For r = 1 To BRows()
        If Abs(CDbl(BVal(r, 1)) - d) < 0.001 _
        And Abs(CDbl(BVal(r, 2)) - t) < 0.001 _
        And Abs(CDbl(BVal(r, 3)) - d1) < 0.001 Then
            TransAutoT1 = CDbl(BVal(r, 4))
            Exit Function
        End If
    Next r
    Err.Raise vbObjectError + 33, , "Нет такого перехода D=" & d & " T=" & t & " D1=" & d1 & " в базе."
End Function

Public Function TransDes(ByVal d As Variant, ByVal t As Variant, ByVal d1 As Variant, _
                         ByVal T1 As Variant, ByVal st As String, ByVal podk As Boolean, _
                         Optional ByVal L As Variant, Optional ByVal lMulti As Boolean) As String
    Dim tk As Object, s As String
    Set tk = CreateObject("Scripting.Dictionary")
    tk("P") = TokPodk(podk)
    tk("D") = NumTxt(d)
    tk("T") = NumTxt(t)
    tk("D1") = NumTxt(d1)
    tk("T1") = NumTxt(T1)
    tk("STSEG") = TokSteelSeg(st)
    s = BuildFromDict("TRANS", tk)
    If lMulti And Not isEmpty(L) Then s = s & ", L=" & NumTxt(L) & "мм"
    TransDes = s
End Function

Public Sub WriteTrans(ByVal tgt As Range, ByVal d As Double, ByVal t As Double, _
                      ByVal d1 As Double, ByVal T1 As Double, ByVal L As Double, _
                      ByVal st As String, ByVal podk As Boolean)
    Dim ws As Worksheet, r As Long, k As Long, h As Long
    Dim ev As Boolean, lMulti As Boolean
    ev = Application.EnableEvents
    On Error GoTo fin
    Set ws = ShSv()
    r = tgt.row
    h = HdrOf(r)
    If h = 0 Then Err.Raise vbObjectError + 30, , "Строка " & r & " не входит в блок позиций."
    Application.EnableEvents = False

    k = TransRow(d, t, d1, T1, L)
    If k = 0 Then Err.Raise vbObjectError + 32, , "Нет такого перехода в базе tTransitions."
    lMulti = (TransValues(6, d, t, d1, T1).Count > 1)
    UseBase "TRANS"

Public Function FlangeValues(ByVal field As Long, ByVal dn As Variant, _
                             ByVal pn As Variant, ByVal typ As Variant) As Collection
    Dim r As Long, c As New Collection, v As Variant, ok As Boolean
    UseBase "FLANGE"
    For r = 1 To BRows()
        ok = True
        If Not isEmpty(dn) Then ok = (Abs(CDbl(BVal(r, 1)) - CDbl(dn)) < 0.001)
        If ok And Not isEmpty(pn) Then ok = (Abs(CDbl(BVal(r, 2)) - CDbl(pn)) < 0.001)
        If ok And Not isEmpty(typ) Then ok = (Abs(CDbl(BVal(r, 3)) - CDbl(typ)) < 0.001)
        If ok Then
            v = BVal(r, field)
            On Error Resume Next
            c.Add v, CStr(v)
            On Error GoTo 0
        End If
    Next r
    Set FlangeValues = c
End Function

Public Function FlangeRow(ByVal dn As Double, ByVal pn As Double, ByVal typ As String) As Long
    Dim r As Long
    UseBase "FLANGE"
    For r = 1 To BRows()
        If Abs(CDbl(BVal(r, 1)) - dn) < 0.001 _
        And Abs(CDbl(BVal(r, 2)) - pn) < 0.001 _
        And Abs(CDbl(BVal(r, 3)) - CDbl(typ)) < 0.001 Then
            FlangeRow = r: Exit Function
        End If
    Next r
    FlangeRow = 0
End Function

' ==================== ОПОРЫ ====================
Public Function SupportExec(ByVal baseExec As String, ByVal welded As Boolean, _
                            ByVal holes As Boolean, ByVal waste As Boolean) As String
    Dim s As String
    s = baseExec
    If welded Then
        s = Replace(s, "А", "АС")
        s = Replace(s, "Б", "БС")
    End If
    If holes Then s = s & "О"
    If waste Then s = s & "в"
    SupportExec = s
End Function

Public Function SupportKey(ByVal sCode As String, ByVal d As Variant, _
                           ByVal exec As String, ByVal fixed As Boolean) As String
    Dim s As String, sh As String
    If Left$(sCode, 4) <> "SUP_" Then _
        Err.Raise vbObjectError + 34, , "Код опоры должен начинаться с SUP_: " & sCode
    sh = Mid$(sCode, 5)
    s = exec
    If Right$(s, 1) = "в" Then s = Left$(s, Len(s) - 1)
    If Right$(s, 1) = "О" Then s = Left$(s, Len(s) - 1)
    SupportKey = "SUP|" & sh & "|" & NumKey(d) & "|" & s & IIf(fixed, "|У", "")
End Function

Public Function SupportDes(ByVal sCode As String, ByVal d As Variant, ByVal exec As String, _
                           ByVal Steel As String, ByVal fixed As Boolean) As String
    Dim tk As Object
    Set tk = CreateObject("Scripting.Dictionary")
    tk("D") = NumTxt(d)
    tk("EXEC") = exec
    tk("ST") = Steel
    tk("FIX") = IIf(fixed, " (в комплекте с упором)", "")
    SupportDes = BuildFromDict(sCode, tk)
End Function

Private Function CharCodes(ByVal s As String) As String
    Dim i As Long, msg As String
    For i = 1 To Len(s)

    Next i
    CharCodes = msg
End Function

Public Function ValveMass(ByVal code As String, ByVal dn As Double, ByVal pn As Double, _
ByVal params As String, ByVal upl As String) As Double
Dim baseMass As Double, k As Long, totalMass As Double
Dim p() As String
Dim zpMass As Double, zfMass As Double, flOneMass As Double, nFl As Long
Dim r As Long
k = ArmRow(code, dn, pn)
If k = 0 Then ValveMass = 0: Exit Function
UseBase code
baseMass = CDbl(BVal(k, 4))
totalMass = baseMass
If Len(Trim$(params)) = 0 Then ValveMass = totalMass: Exit Function
p = Split(params, "|")
If UBound(p) < 17 Then ValveMass = totalMass: Exit Function
If p(7) = "1" Then
UseBase "BLIND_INT"
For r = 1 To BRows()
If CStr(BVal(r, 1)) = "1" And CStr(BVal(r, 2)) = p(8) And _
Abs(CDbl(BVal(r, 3)) - dn) < 0.001 And _
Abs(CDbl(BVal(r, 4)) - pn) < 0.001 Then
zpMass = CDbl(BVal(r, 5))
Exit For
End If
Next r
totalMass = totalMass + zpMass
End If
If p(10) = "1" Then
UseBase "BLIND_FL"
For r = 1 To BRows()
If Abs(CDbl(BVal(r, 1)) - dn) < 0.001 And _
Abs(CDbl(BVal(r, 2)) - pn) < 0.001 And _
CStr(BVal(r, 3)) = "1" Then
zfMass = CDbl(BVal(r, 5))
Exit For
End If
Next r
totalMass = totalMass + zfMass
End If
If p(12) = "1" Then
nFl = IIf(p(10) = "1", 1, 2)
Dim flUpl As String
flUpl = IIf(Left$(upl, 1) = "E", "E", "B")
UseBase "FLANGE"
For r = 1 To BRows()
If Abs(CDbl(BVal(r, 1)) - dn) < 0.001 And _
Abs(CDbl(BVal(r, 2)) - pn) < 0.001 And _
CStr(BVal(r, 3)) = p(13) And _
CStr(BVal(r, 4)) = flUpl Then
flOneMass = CDbl(BVal(r, 6))
Exit For
End If
Next r
totalMass = totalMass + nFl * flOneMass
End If
ValveMass = totalMass
End Function
