    QuizSheet As String
    UnitText As String

' ==================== ТОЧКИ ВХОДА ====================
Public Sub ExportSpecToWord()
    Dim n() As String, t() As String, c As Long
    BuildStampPairs n, t, c
    RunExport n, t, c, True
End Sub

Public Sub ExportStampOnlyToWord()
    Dim n() As String, t() As String, c As Long
    BuildStampPairs n, t, c
    RunExport n, t, c, False
End Sub

Public Sub ExportStampAndSpecToWord()
    ExportSpecToWord
End Sub

' ==================== ДВИЖОК ====================
Private Sub RunExport(ByRef namesArr() As String, ByRef tagsArr() As String, _
                      ByVal cnt As Long, ByVal withTable As Boolean)
    Dim tplPath As String, outPath As String, cipher As String
    Dim wdApp As Object, wdDoc As Object, tbl As Object
    Dim items() As SpecItem, itemCnt As Long
    Dim errNum As Long, errDesc As String
    Dim tAll As Double

    mLog = "": mT = Timer: tAll = Timer
    mStep = "старт"

    tplPath = GetTemplatePath()
    If Len(tplPath) = 0 Then
        MsgBox "Не найден шаблон:" & vbCrLf & _
               ThisWorkbook.Path & "\" & TEMPLATE_SUBFOLDER & "\" & vbCrLf & _
               TEMPLATE_NAME & "  или  " & TEMPLATE_NAME_FALLBACK, vbExclamation
        Exit Sub
    End If
    Tick "проверка шаблона"

    On Error GoTo eh

    If withTable Then
        mStep = "сбор позиций"
        CollectAndAggregateSpec items, itemCnt
        Tick "сбор + суммирование"
        If itemCnt > 1 Then
            mStep = "сортировка"
            SortSpecItems items, itemCnt
            Tick "сортировка"
        End If
    End If

    mStep = "шифр / имя файла"
    cipher = TryGetNamedRangeText(namesArr, tagsArr, cnt, tagsArr(1))
    outPath = OutFolder() & SafeFileName(cipher) & ".docx"

    mStep = "запуск Word"
    Set wdApp = CreateWordWithRetry()
    Tick "запуск Word"

    WordSpeedOn wdApp
    wdApp.Visible = False

    mStep = "документ из шаблона"
    Set wdDoc = NewDocFromTemplate(wdApp, tplPath)
    Tick "открытие шаблона"

    mStep = "переменные штампа + поля"
    ApplyStampTagsToWord wdDoc, namesArr, tagsArr, cnt

    If withTable Then
        mStep = "поиск таблицы спецификации"
        Set tbl = FindSpecTable(wdDoc)
        If tbl Is Nothing Then
            Err.Raise vbObjectError + 101, "RunExport", _
                      "Не найдена таблица спецификации (9 колонок с заголовком «Наименование»)."
        End If

        mStep = "заполнение таблицы"
        FillWordSpecTable tbl, items, itemCnt
        Tick "таблица: строки + ячейки"

        mStep = "оформление таблицы"
        FormatSpecTable tbl, wdApp
        Tick "таблица: оформление"

        mStep = "финальное обновление полей"
        UpdateFieldsEverywhere wdDoc
        Tick "поля: финальное обновление"
    End If

    mStep = "возврат настроек Word"
    WordSpeedOff wdApp

    mStep = "сохранение"
    SaveDocWithRetry wdDoc, outPath
    Tick "сохранение"

    mLog = mLog & "ИТОГО: " & Format$(Timer - tAll, "0.00") & " s" & vbCrLf & _
           "Позиций: " & itemCnt
    DumpTiming

    wdApp.Visible = True
    On Error Resume Next
    wdDoc.Activate
    On Error GoTo 0

    MsgBox "Готово: " & outPath & vbCrLf & vbCrLf & mLog, vbInformation
    Exit Sub

eh:
    errNum = Err.Number: errDesc = Err.Description
    On Error Resume Next
    WordSpeedOff wdApp
    If Not wdDoc Is Nothing Then wdDoc.Close SaveChanges:=False
    If Not wdApp Is Nothing Then wdApp.Quit
    mLog = mLog & "СБОЙ НА ШАГЕ: " & mStep
    DumpTiming
    On Error GoTo 0
    MsgBox "Ошибка " & errNum & vbCrLf & "ШАГ: " & mStep & vbCrLf & vbCrLf & errDesc, vbExclamation
End Sub

' ==================== ШТАМП: ПЕРЕМЕННЫЕ + ПОЛЯ ====================
Private Sub ApplyStampTagsToWord(ByVal wdDoc As Object, ByRef namesArr() As String, _
                                 ByRef tagsArr() As String, ByVal cnt As Long)
    Dim i As Long, v As String, nSet As Long, nMiss As Long
    Dim miss As String

    For i = 1 To cnt
        If Len(namesArr(i)) > 0 Then
            If TryReadRangeText(namesArr(i), v) Then
                SetDocVar wdDoc, namesArr(i), v
                nSet = nSet + 1
            Else
                nMiss = nMiss + 1
                miss = miss & namesArr(i) & " "
            End If
        End If
    Next i

    UpdateFieldsEverywhere wdDoc

    Tick "штамп: переменных " & nSet & "; полей тело/колонтитулы/фигуры " & _
         mFldDoc & "/" & mFldHF & "/" & mFldShp
    If nMiss > 0 Then Tick "штамп: НЕТ диапазонов (" & nMiss & "): " & miss
End Sub

Private Sub SetDocVar(ByVal wdDoc As Object, ByVal nm As String, ByVal val As String)
    On Error Resume Next
    wdDoc.Variables.Add nm, val
    If Err.Number <> 0 Then
        Err.Clear
        wdDoc.Variables(nm).Value = val
    End If
    Err.Clear
End Sub

Private Sub UpdateFieldsEverywhere(ByVal wdDoc As Object)
    Dim sec As Object, shp As Object
    mFldDoc = 0: mFldHF = 0: mFldShp = 0
    On Error Resume Next
    mFldDoc = wdDoc.Fields.Count
    wdDoc.Fields.Update
    For Each sec In wdDoc.Sections
        UpdHF sec.Headers(2)
        UpdHF sec.Headers(1)
        UpdHF sec.Headers(3)
        UpdHF sec.Footers(2)
        UpdHF sec.Footers(1)
        UpdHF sec.Footers(3)
    Next sec
    For Each shp In wdDoc.Shapes
        UpdShape shp
    Next shp
End Sub

Private Sub UpdHF(ByVal hf As Object)
    Dim shp As Object
    If hf Is Nothing Then Exit Sub
    On Error Resume Next
    If Not hf.exists Then Exit Sub
    mFldHF = mFldHF + hf.Range.Fields.Count
    hf.Range.Fields.Update
    For Each shp In hf.Shapes
        UpdShape shp
    Next shp
End Sub

Private Sub UpdShape(ByVal shp As Object)
    Dim subShp As Object, tr As Object
    If shp Is Nothing Then Exit Sub
    On Error Resume Next
    If shp.Type = 6 Then
        For Each subShp In shp.GroupItems
            UpdShape subShp
        Next subShp
    Else
        If shp.HasTextFrame Then
            If shp.TextFrame.HasText Then
                Set tr = shp.TextFrame.TextRange
                mFldShp = mFldShp + tr.Fields.Count
                tr.Fields.Update
            End If
        End If
    End If
End Sub

' ==================== ЗАМЕРЫ ====================
Private Sub Tick(ByVal label As String)
    mLog = mLog & label & ": " & Format$(Timer - mT, "0.00") & " s" & vbCrLf
    mT = Timer
End Sub

Private Sub DumpTiming()
    Dim f As Integer
    On Error Resume Next
    f = FreeFile
    Open OutFolder() & "_word_timing.txt" For Output As #f
    Print #f, mLog
    Close #f
End Sub

' ==================== УСКОРЕНИЕ WORD ====================
Private Sub WordSpeedOn(ByVal wdApp As Object)
    If wdApp Is Nothing Then Exit Sub
    On Error Resume Next
    With wdApp
        mScr = .ScreenUpdating
        .ScreenUpdating = False
        .DisplayAlerts = 0
        mSpell = .Options.CheckSpellingAsYouType
        .Options.CheckSpellingAsYouType = False
        mGram = .Options.CheckGrammarAsYouType
        .Options.CheckGrammarAsYouType = False
        mBg = .Options.BackgroundSave
        .Options.BackgroundSave = False
    End With
    On Error GoTo 0
End Sub

Private Sub WordSpeedOff(ByVal wdApp As Object)
    If wdApp Is Nothing Then Exit Sub
    On Error Resume Next
    With wdApp
        .Options.CheckSpellingAsYouType = mSpell
        .Options.CheckGrammarAsYouType = mGram
        .Options.BackgroundSave = mBg
        .ScreenUpdating = mScr
    End With
    On Error GoTo 0
End Sub

' ==================== ШАБЛОН / WORD ====================
Private Function GetTemplatePath() As String
    Dim base As String, p As String
    base = ThisWorkbook.Path & "\" & TEMPLATE_SUBFOLDER & "\"
    p = base & TEMPLATE_NAME
    If Len(Dir$(p)) > 0 Then
        GetTemplatePath = p
        Exit Function
    End If
    p = base & TEMPLATE_NAME_FALLBACK
    If Len(Dir$(p)) > 0 Then GetTemplatePath = p
End Function

Private Function NewDocFromTemplate(ByVal wdApp As Object, ByVal tplPath As String) As Object
    Dim attempt As Long, docObj As Object
    Dim isDotx As Boolean
    isDotx = (LCase$(Right$(tplPath, 5)) = ".dotx")

    For attempt = 1 To 10
        Err.Clear
        On Error Resume Next
        If isDotx Then
            Set docObj = wdApp.Documents.Add(Template:=tplPath, NewTemplate:=False, DocumentType:=0)
        Else
            Set docObj = wdApp.Documents.Open(fileName:=tplPath, ReadOnly:=True, AddToRecentFiles:=False)
        End If
        If Err.Number = 0 And Not docObj Is Nothing Then
            On Error GoTo 0
            Set NewDocFromTemplate = docObj
            Exit Function
        End If
        If Err.Number <> -2147418110 Then
            On Error GoTo 0
            Err.Raise Err.Number, Err.Source, Err.Description
        End If
        On Error GoTo 0
        DoEvents
        Sleep 60
    Next attempt
    Err.Raise vbObjectError + 102, "NewDocFromTemplate", "Не удалось создать документ из шаблона."
End Function

Private Function CreateWordWithRetry() As Object
    Dim attempt As Long, appObj As Object
    For attempt = 1 To 10
        Err.Clear
        On Error Resume Next
        Set appObj = CreateObject("Word.Application")
        If Err.Number = 0 And Not appObj Is Nothing Then
            On Error GoTo 0
            Set CreateWordWithRetry = appObj
            Exit Function
        End If
        If Err.Number <> -2147418110 Then
            On Error GoTo 0
            Err.Raise Err.Number, Err.Source, Err.Description
        End If
        On Error GoTo 0
        DoEvents
        Sleep 60
    Next attempt
    Err.Raise -2147418110, "CreateWordWithRetry", "Не удалось запустить Word."
End Function

Private Sub SaveDocWithRetry(ByVal wdDoc As Object, ByVal outPath As String)
    Dim attempt As Long
    For attempt = 1 To 10
        Err.Clear
        On Error Resume Next
        wdDoc.SaveAs2 fileName:=outPath, FileFormat:=12, AddToRecentFiles:=False
        If Err.Number = 0 Then
            On Error GoTo 0
            Exit Sub
        End If
        If Err.Number <> -2147418110 Then
            On Error GoTo 0
            Err.Raise Err.Number, Err.Source, Err.Description
        End If
        On Error GoTo 0
        DoEvents
        Sleep 60
    Next attempt
    Err.Raise -2147418110, "SaveDocWithRetry", "Не удалось сохранить документ Word."
End Sub

' ==================== ТАБЛИЦА СПЕЦИФИКАЦИИ ====================
Private Function FindSpecTable(ByVal wdDoc As Object) As Object
    Dim tbls As Object, tbl As Object, trng As Object
    Dim n As Long, txt As String
    On Error Resume Next
    Set tbls = wdDoc.Tables
    For Each tbl In tbls
        n = 0
        n = tbl.Columns.Count
        Err.Clear
        If n = 9 Then
            Set trng = tbl.Range
            txt = Left$(trng.Text, 300)
            If InStr(1, txt, "Наименование", vbTextCompare) > 0 Then
                Set FindSpecTable = tbl
                Exit Function
            End If
        End If
    Next tbl
End Function

Private Sub FillWordSpecTable(ByVal tbl As Object, ByRef items() As SpecItem, ByVal itemCnt As Long)
    Const FIRST_DATA_ROW As Long = 2
    Dim nRows As Long, keep As Long, i As Long, rowIdx As Long
    Dim rng As Object

    nRows = tbl.Rows.Count
    keep = FIRST_DATA_ROW + itemCnt - 1
    If keep < FIRST_DATA_ROW Then keep = FIRST_DATA_ROW

    Do While nRows < keep
        tbl.Rows.Add
        nRows = nRows + 1
    Loop

    If nRows > keep Then
        On Error Resume Next
        Set rng = tbl.Rows(keep + 1).Range
        rng.End = tbl.Range.End
        rng.Rows.Delete
        On Error GoTo 0

        nRows = tbl.Rows.Count
        Do While nRows > keep
            tbl.Rows(nRows).Delete
            nRows = nRows - 1
        Loop
    End If

    For i = 1 To itemCnt
        rowIdx = FIRST_DATA_ROW + i - 1
        SetCell tbl, rowIdx, 2, items(i).NameText
        SetCell tbl, rowIdx, 3, items(i).QuizSheet
        SetCell tbl, rowIdx, 6, items(i).UnitText
        If items(i).HasQty Then SetCell tbl, rowIdx, 7, FormatNumberClean(items(i).QtySum)
        SetCell tbl, rowIdx, 8, items(i).UnitMassText
        SetCell tbl, rowIdx, 9, items(i).NoteText
    Next i
End Sub

Private Sub SetCell(ByVal tbl As Object, ByVal r As Long, ByVal c As Long, ByVal s As String)
    Dim clRng As Object
    If Len(s) = 0 Then Exit Sub
    On Error Resume Next
    Set clRng = tbl.cell(r, c).Range
    If clRng Is Nothing Then Exit Sub
    clRng.MoveEnd 1, -1
    clRng.Text = s
End Sub

Private Sub FormatSpecTable(ByVal tbl As Object, ByVal wdApp As Object, _
                            Optional ByVal applyAlign As Boolean = False)
    On Error Resume Next
    With tbl
        .Rows.HeightRule = 1
        .Rows.Height = wdApp.CentimetersToPoints(0.8)
        .Rows.AllowBreakAcrossPages = False
        .Rows(1).HeadingFormat = True
    End With

    If applyAlign Then
        With tbl
            .Range.ParagraphFormat.SpaceBefore = 0
            .Range.ParagraphFormat.SpaceAfter = 0
            .Range.ParagraphFormat.LineSpacingRule = 0
            .Columns(2).Range.ParagraphFormat.Alignment = 0
            .Columns(2).Cells.VerticalAlignment = 1
            .Columns(3).Range.ParagraphFormat.Alignment = 0
            .Columns(3).Cells.VerticalAlignment = 1
            .Columns(6).Range.ParagraphFormat.Alignment = 1
            .Columns(6).Cells.VerticalAlignment = 1
            .Columns(7).Range.ParagraphFormat.Alignment = 1
            .Columns(7).Cells.VerticalAlignment = 1
            .Columns(8).Range.ParagraphFormat.Alignment = 2
            .Columns(8).Cells.VerticalAlignment = 1
            .Columns(9).Range.ParagraphFormat.Alignment = 0
            .Columns(9).Cells.VerticalAlignment = 1
        End With
    End If
End Sub

' ==================== СБОР ПОЗИЦИЙ ====================
Private Sub CollectAndAggregateSpec(ByRef items() As SpecItem, ByRef itemCnt As Long)
    Dim ws As Worksheet
    Dim r As Long, lastR As Long, h As Long, mk As String
    Dim cName As Long, cQuiz As Long, cUnit As Long, cQty As Long, cMass1 As Long, cNote As Long
    Dim nm As String, qz As String, un As String, m1 As String, nt As String
    Dim qtyVal As Double, hasQ As Boolean
    Dim idx As Long, foundIdx As Long
    Dim f As Range

    itemCnt = 0
    Set ws = ThisWorkbook.Worksheets("Сводная")
    Set f = ws.Cells.Find(What:="*", LookIn:=xlFormulas, _
                          SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
    If f Is Nothing Then Exit Sub
    lastR = f.row
    If lastR < 1 Then Exit Sub

    For r = 1 To lastR
        mk = UCase$(Trim$(CStr(ws.Cells(r, 9).Value)))
        If mk = "HDR" Then
            h = r
            FindSpecColumns ws, h, cName, cQuiz, cUnit, cQty, cMass1, cNote
        ElseIf mk <> "SEC" And mk <> "GAP" And h > 0 And r > h Then
            If cName > 0 Then
                nm = CStr(ws.Cells(r, cName).Value)
                If Len(nm) > 0 Then
                    qz = ReadColStr(ws, r, cQuiz)
                    un = ReadColStr(ws, r, cUnit)
                    m1 = ReadColStr(ws, r, cMass1)
                    nt = ReadColStr(ws, r, cNote)
                    ReadColQty ws, r, cQty, qtyVal, hasQ

                    foundIdx = FindMatchingSpecItem(items, itemCnt, nm, qz, un, m1, nt)
                    If foundIdx > 0 Then
                        If hasQ Then
                            items(foundIdx).QtySum = items(foundIdx).QtySum + qtyVal
                            items(foundIdx).HasQty = True
                        End If
                    Else
                        itemCnt = itemCnt + 1
                        If itemCnt = 1 Then
                            ReDim items(1 To 32)
                        ElseIf itemCnt > UBound(items) Then
                            ReDim Preserve items(1 To UBound(items) * 2)
                        End If

                        idx = itemCnt
                        items(idx).NameText = nm
                        items(idx).QuizSheet = qz
                        items(idx).UnitText = un
                        items(idx).QtySum = IIf(hasQ, qtyVal, 0#)
                        items(idx).HasQty = hasQ
                        items(idx).UnitMassText = m1
                        items(idx).NoteText = nt
                        items(idx).SortWeight = GetWeightSafe(r)
                        items(idx).SortDia = GetDiaSafe(r)
                        items(idx).SortThk = GetThkSafe(r)
                        items(idx).SortPn = GetPnSafe(r)
                        items(idx).OrigIndex = idx
                    End If
                End If
            End If
        End If
    Next r
End Sub

Private Sub FindSpecColumns(ByVal ws As Worksheet, ByVal hdrRow As Long, _
                            ByRef cName As Long, ByRef cQuiz As Long, ByRef cUnit As Long, _
                            ByRef cQty As Long, ByRef cMass1 As Long, ByRef cNote As Long)
    Dim c As Long, hTxt As String
    cName = 0: cQuiz = 0: cUnit = 0: cQty = 0: cMass1 = 0: cNote = 0

    For c = 1 To 40
        If c <> 9 Then
            hTxt = Trim$(CStr(ws.Cells(hdrRow, c).Value))
            Select Case hTxt
                Case "Наименование": If cName = 0 Then cName = c
                Case "Опросной лист", "Опросный лист": If cQuiz = 0 Then cQuiz = c
                Case "Ед. изм.", "Ед.изм.": If cUnit = 0 Then cUnit = c
                Case "Кол-во": If cQty = 0 Then cQty = c
                Case "Масса 1 ед": If cMass1 = 0 Then cMass1 = c
                Case "Примечание": If cNote = 0 Then cNote = c
            End Select
        End If
    Next c

    If cName = 0 Then cName = 2
    If cQuiz = 0 Then cQuiz = 3
    If cUnit = 0 Then cUnit = 4
    If cQty = 0 Then cQty = 5
    If cMass1 = 0 Then cMass1 = 6
    If cNote = 0 Then cNote = 8
End Sub

Private Function ReadColStr(ByVal ws As Worksheet, ByVal r As Long, ByVal c As Long) As String
    If c <= 0 Then Exit Function
    ReadColStr = CStr(ws.Cells(r, c).Value)
End Function

Private Sub ReadColQty(ByVal ws As Worksheet, ByVal r As Long, ByVal c As Long, _
                       ByRef outQty As Double, ByRef outHasQty As Boolean)
    Dim v As Variant, s As String
    outQty = 0#
    outHasQty = False
    If c <= 0 Then Exit Sub

    v = ws.Cells(r, c).Value
    If isEmpty(v) Then Exit Sub
    If IsNumeric(v) Then
        outQty = CDbl(v)
        outHasQty = True
        Exit Sub
    End If

    s = Trim$(Replace(CStr(v), ",", "."))
    If Len(s) > 0 And IsNumeric(s) Then
        outQty = CDbl(s)
        outHasQty = True
    End If
End Sub

Private Function FindMatchingSpecItem(ByRef items() As SpecItem, ByVal itemCnt As Long, _
                                      ByVal nm As String, ByVal qz As String, _
                                      ByVal un As String, ByVal m1 As String, _
                                      ByVal nt As String) As Long
    Dim i As Long
    For i = 1 To itemCnt
        If StrComp(items(i).NameText, nm, vbBinaryCompare) = 0 Then
            If StrComp(items(i).QuizSheet, qz, vbBinaryCompare) = 0 Then
                If StrComp(items(i).UnitText, un, vbBinaryCompare) = 0 Then
                    If StrComp(items(i).UnitMassText, m1, vbBinaryCompare) = 0 Then
                        If StrComp(items(i).NoteText, nt, vbBinaryCompare) = 0 Then
                            FindMatchingSpecItem = i
                            Exit Function
                        End If
                    End If
                End If
            End If
        End If
    Next i
    FindMatchingSpecItem = 0
End Function

' ==================== СОРТИРОВКА ====================
Private Function GetWeightSafe(ByVal r As Long) As Double
    GetWeightSafe = 999999#
    On Error Resume Next
    GetWeightSafe = modLayout.GetSortWeight(r)
    If Err.Number <> 0 Then Err.Clear: GetWeightSafe = 999999#
    On Error GoTo 0
End Function

Private Function GetDiaSafe(ByVal r As Long) As Double
    GetDiaSafe = 0#
    On Error Resume Next
    GetDiaSafe = modLayout.GetSortDia(r)
    If Err.Number <> 0 Then Err.Clear: GetDiaSafe = 0#
    On Error GoTo 0
End Function

Private Function GetThkSafe(ByVal r As Long) As Double
    GetThkSafe = 0#
    On Error Resume Next
    GetThkSafe = modLayout.GetSortThk(r)
    If Err.Number <> 0 Then Err.Clear: GetThkSafe = 0#
    On Error GoTo 0
End Function

Private Function GetPnSafe(ByVal r As Long) As Double
    GetPnSafe = 0#
    On Error Resume Next
    GetPnSafe = modLayout.GetSortPn(r)
    If Err.Number <> 0 Then Err.Clear: GetPnSafe = 0#
    On Error GoTo 0
End Function

Private Function SpecItemComesBefore(ByRef a As SpecItem, ByRef b As SpecItem) As Boolean
    If a.SortWeight <> b.SortWeight Then
        SpecItemComesBefore = (a.SortWeight < b.SortWeight)
        Exit Function
    End If
    If a.SortDia <> b.SortDia Then
        SpecItemComesBefore = (a.SortDia > b.SortDia)
        Exit Function
    End If
    If a.SortThk <> b.SortThk Then
        SpecItemComesBefore = (a.SortThk > b.SortThk)
        Exit Function
    End If
    If a.SortPn <> b.SortPn Then
        SpecItemComesBefore = (a.SortPn > b.SortPn)
        Exit Function
    End If
    SpecItemComesBefore = (a.OrigIndex < b.OrigIndex)
End Function

Private Sub SortSpecItems(ByRef items() As SpecItem, ByVal itemCnt As Long)
    Dim i As Long, j As Long, tmp As SpecItem
    For i = 1 To itemCnt - 1
        For j = i + 1 To itemCnt
            If SpecItemComesBefore(items(j), items(i)) Then
                tmp = items(i)
                items(i) = items(j)
                items(j) = tmp
            End If
        Next j
    Next i
End Sub

' ==================== ИМЕНОВАННЫЕ ДИАПАЗОНЫ ====================
Private Sub AddPair(ByRef namesArr() As String, ByRef tagsArr() As String, _
                    ByRef cnt As Long, ByVal nmRange As String, ByVal tagWord As String)
    cnt = cnt + 1
    If cnt > UBound(namesArr) Then
        ReDim Preserve namesArr(1 To cnt + 16)
        ReDim Preserve tagsArr(1 To cnt + 16)
    End If
    namesArr(cnt) = Trim$(nmRange)
    tagsArr(cnt) = Trim$(tagWord)
End Sub

Private Function TryReadRangeText(ByVal nmRange As String, ByRef outText As String) As Boolean
    Dim nm As Name
    Dim rng As Range

    outText = ""
    On Error Resume Next
    Set nm = ThisWorkbook.names(nmRange)
    If Not nm Is Nothing Then Set rng = nm.RefersToRange
    If rng Is Nothing Then Set rng = ThisWorkbook.Range(nmRange)
    If rng Is Nothing Then Exit Function
    outText = Trim$(CStr(rng.Cells(1, 1).Text))
    If Err.Number <> 0 Then
        Err.Clear
        Exit Function
    End If
    On Error GoTo 0
    TryReadRangeText = True
End Function

Private Function TryGetNamedRangeText(ByRef namesArr() As String, ByRef tagsArr() As String, _
                                      ByVal cnt As Long, ByVal targetTag As String) As String
    Dim i As Long, txt As String
    For i = 1 To cnt
        If UCase$(tagsArr(i)) = UCase$(targetTag) Then
            If TryReadRangeText(namesArr(i), txt) Then TryGetNamedRangeText = txt
            Exit Function
        End If
    Next i
End Function

Private Function FormatNumberClean(ByVal d As Double) As String
    If Abs(d - Round(d, 0)) < 0.00000001 Then
        FormatNumberClean = CStr(CLng(Round(d, 0)))
    Else
        FormatNumberClean = CStr(Round(d, 4))
    End If
End Function

Private Function OutFolder() As String
    Dim p As String
    p = ThisWorkbook.Path
    If Len(p) = 0 Then p = Environ$("USERPROFILE") & "\Documents"
    If Right$(p, 1) <> "\" Then p = p & "\"
    OutFolder = p
End Function

Private Function SafeFileName(ByVal s As String) As String
    Dim bad As Variant, i As Long
    s = Replace(Replace(Trim$(s), vbLf, " "), vbCr, " ")
    bad = Array("/", "\", ":", "*", "?", """", "<", ">", "|")
    For i = LBound(bad) To UBound(bad)
        s = Replace(s, CStr(bad(i)), "_")
    Next i
    If Len(s) = 0 Then s = "Документ"
    If Len(s) > 150 Then s = Left$(s, 150)
    SafeFileName = s
End Function
