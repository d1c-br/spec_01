Private Function GetForm(ByVal nm As String) As Object
    Dim proj As Object, comp As Object, i As Long
    Set proj = ThisWorkbook.VBProject
    For i = 1 To proj.VBComponents.Count
        If LCase$(proj.VBComponents(i).Name) = LCase$(nm) Then
            Set GetForm = proj.VBComponents(i)
            Exit Function
        End If
    Next i
    Set comp = proj.VBComponents.Add(3)   '3 = UserForm
    On Error GoTo renameFail
    comp.Name = nm
    On Error GoTo 0
    Set GetForm = comp
    Exit Function
renameFail:
    Err.Raise vbObjectError + 50, , "Не удалось переименовать новую форму в «" & nm & _
        "». Скорее всего, в проекте уже есть компонент с таким именем, или открыто окно дизайнера. " & _
        "Закройте окна форм в VBE, удалите дубликат и повторите."
End Function

Private Sub ClearControls(ByVal comp As Object)
    Do While comp.Designer.Controls.Count > 0
        comp.Designer.Controls.Remove 0
    Loop
End Sub

Private Sub AddLabel(ByVal comp As Object, ByVal nm As String, ByVal cap As String, _
                     ByVal L As Single, ByVal t As Single, ByVal w As Single, ByVal h As Single, _
                     Optional ByVal border As Long = 0)
    Dim c As Object
    Set c = comp.Designer.Controls.Add("Forms.Label.1")
    c.Name = nm: c.Caption = cap
    c.Left = L: c.Top = t: c.Width = w: c.Height = h
    c.BorderStyle = border
End Sub
