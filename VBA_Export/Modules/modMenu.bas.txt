Option Explicit

Private Const MENU_TAG As String = "SvodMenu"
Private Const CB_POPUP As Long = 10    ' msoControlPopup
Private Const CB_BUTTON As Long = 1    ' msoControlButton

' ---------- доступ к нашему подменю ----------
Public Function МенюПопап() As Object
    Dim bar As CommandBar, Ctl As Object
    On Error Resume Next
    Set bar = Application.CommandBars("Cell")
    On Error GoTo 0
    If bar Is Nothing Then Exit Function
    For Each Ctl In bar.Controls
        If CStr(Ctl.tag) = MENU_TAG Then
            Set МенюПопап = Ctl
            Exit Function
        End If
    Next Ctl
End Function

Public Sub УдалитьКонтекстноеМеню()
    Dim bar As CommandBar, i As Long
    On Error Resume Next
    Set bar = Application.CommandBars("Cell")
    If bar Is Nothing Then Exit Sub
    For i = bar.Controls.Count To 1 Step -1   ' обратный цикл: удаление во время перебора
        If CStr(bar.Controls(i).tag) = MENU_TAG Then bar.Controls(i).Delete
    Next i
    On Error GoTo 0
End Sub

Public Sub МенюПрименитьЗону(ByVal z As String, Optional ByVal tempState As String = "")
Dim pop As Object, Ctl As Object, tags As String, cTemp As Object
Dim arr As Variant, i As Long
Set pop = МенюПопап()
If pop Is Nothing Then Exit Sub
For Each Ctl In pop.Controls
Ctl.Visible = False
Next Ctl
Select Case z
Case "SEC": tags = "tSecEdit,tSecAbove,tSecBelow,tSecDel,tSort,tSortAll,tCalc,tBorders"
Case "HDR": tags = "tSecEdit,tPosEdit,tPosAddAbove,tPosAdd,tRowInsAbove,tRowIns,tPosClear,tRowDel,tSort,tSortAll,tCalc,tBorders"
Case "POS": tags = "tPosEdit,tPosAddAbove,tPosAdd,tRowInsAbove,tRowIns,tPosClear,tRowDel,tSort,tSortAll,tCalc,tBorders"
Case "GAP": tags = "tGapAdd,tSortAll,tCalc,tBorders"
Case Else:  tags = "tFreeAdd,tSortAll,tCalc,tBorders"
End Select
arr = Split(tags, ",")
For Each Ctl In pop.Controls
For i = 0 To UBound(arr)
If CStr(Ctl.tag) = arr(i) Then
Ctl.Visible = True
Exit For
End If
Next i
Next Ctl
For Each Ctl In pop.Controls
If CStr(Ctl.tag) = "tTemp" Then Set cTemp = Ctl: Exit For
Next Ctl
If Not cTemp Is Nothing Then
If z = "POS" And Len(tempState) > 0 Then
cTemp.Visible = True
If tempState = "SENT" Then
cTemp.Caption = "Уже есть во временной базе"
cTemp.Enabled = False
Else
cTemp.Caption = "Добавить во временную базу…"
cTemp.Enabled = True
End If
Else
cTemp.Visible = False
End If
End If
End Sub

' ---------- обработчики пунктов ----------
Public Sub MenuSecEdit()
    Dim s As Long
    s = modLayout.SecOfRow(ActiveCell.row)
    If s = 0 Then MsgBox "Курсор не на описании участка.", vbExclamation: Exit Sub
    frmSection.ShowFor ActiveCell
End Sub

Public Sub MenuSecAbove()
    Dim s As Long
    s = modLayout.SecOfRow(ActiveCell.row)
    If s = 0 Then MsgBox "Курсор не на описании участка.", vbExclamation: Exit Sub
    modLayout.ВставитьБлок s - 1
End Sub

Public Sub MenuSecBelow()
    Dim s As Long, e As Long
    s = modLayout.SecOfRow(ActiveCell.row)
    If s = 0 Then MsgBox "Курсор не на описании участка.", vbExclamation: Exit Sub
    e = modLayout.BlockEnd(s + 4)
    modLayout.ВставитьБлок e + 1
End Sub

Public Sub MenuSecDel()
    Dim s As Long
    s = modLayout.SecOfRow(ActiveCell.row)
    If s = 0 Then MsgBox "Курсор не на описании участка.", vbExclamation: Exit Sub
    modLayout.УдалитьУчасток s
End Sub

Public Sub MenuPosEdit()
    Dim h As Long, k As String
    h = modLayout.HdrOf(ActiveCell.row)
    If h = 0 Then MsgBox "Курсор не на строке позиции.", vbExclamation: Exit Sub
    k = CStr(ActiveCell.Worksheet.Cells(ActiveCell.row, modLayout.col("Тип")).Value)
    If Len(k) > 0 Then modEngine.OpenByType k, ActiveCell Else modEngine.PickAndOpen ActiveCell
End Sub

Public Sub MenuPosAddAbove()
    Dim r As Long
    r = ActiveCell.row

Public Sub MenuPosAdd()

Public Function СпроситьКоличествоСтрок() As Long
Static lastN As Long
Dim s As String, v As Double
If lastN < 1 Then lastN = 1
Do
s = InputBox("Сколько пустых строк вставить?" & vbCrLf & _
"Допустимо от 1 до 100.", "Количество строк", CStr(lastN))
If StrPtr(s) = 0 Then Exit Function
s = Trim$(s)
If Len(s) = 0 Then Exit Function
If IsNumeric(Replace(s, ",", ".")) Then
v = CDbl(Replace(s, ",", "."))
If v = Int(v) And v >= 1 And v <= 100 Then
lastN = CLng(v)
СпроситьКоличествоСтрок = lastN
Exit Function
End If
End If
MsgBox "Нужно целое число от 1 до 100.", vbExclamation, "Количество строк"
Loop
End Function

Public Sub MenuSort()
Dim r As Long, h As Long, s As Long
r = ActiveCell.row
h = modLayout.HdrOf(r)
If h = 0 Then
s = modLayout.SecOfRow(r)
If s > 0 Then h = s + 4
End If
If h = 0 Then MsgBox "Курсор не внутри участка.", vbExclamation: Exit Sub
modLayout.СортироватьУчасток h
End Sub

End Sub

Public Sub MenuTempAdd()
Dim ws As Worksheet, r As Long, h As Long, s As Long, cB As Long
Dim nm As String, un As String, massV As Variant
Dim miss As String, sec As String, newNm As String
Set ws = modLayout.ShSv()
r = ActiveCell.row
h = modLayout.HdrOf(r)
If h = 0 Then MsgBox "Курсор не на строке позиции.", vbExclamation: Exit Sub
If Len(modLayout.СостояниеВнеБазы(r)) = 0 Then
MsgBox "Эта позиция уже в базе — добавление во временную базу не требуется.", vbInformation
Exit Sub
End If
cB = modLayout.col("Наименование")
nm = Trim$(CStr(ws.Cells(r, cB).Value))
un = Trim$(CStr(ws.Cells(r, modLayout.col("Ед.изм.")).Value))
massV = ws.Cells(r, modLayout.col("Масса 1 ед")).Value
If Len(nm) = 0 Then miss = miss & vbCrLf & "  - Наименование"
If isEmpty(massV) Or Not IsNumeric(massV) Then
miss = miss & vbCrLf & "  - Масса 1 ед (числом)"
ElseIf CDbl(massV) <= 0 Then
miss = miss & vbCrLf & "  - Масса 1 ед (больше нуля)"
End If
If Len(un) = 0 Then miss = miss & vbCrLf & "  - Ед.изм."
If Len(miss) > 0 Then
MsgBox "Заполните ячейки:" & miss, vbExclamation, "Не хватает данных"
Exit Sub
End If
If modTemp.TempContains(nm, un) Then
MsgBox "Это наименование уже есть во временной базе.", vbInformation
modLayout.ПодсветитьДублиУчастка h
Exit Sub
End If
newNm = InputBox("Проверьте наименование перед занесением во временную базу." & vbCrLf & _
"При необходимости исправьте текст и нажмите ОК.", "Проверка наименования", nm)
If StrPtr(newNm) = 0 Then Exit Sub
If Len(Trim$(newNm)) = 0 Then Exit Sub
If Trim$(newNm) <> nm Then
ws.Cells(r, cB).Value = Trim$(newNm)
MsgBox "Наименование исправлено. Во временную базу не добавляем." & vbCrLf & vbCrLf & _
"Если это стандартный элемент — выберите его через форму (двойной клик по ячейке).", vbInformation
Exit Sub
End If
s = modLayout.SecOfHdr(h)
If s > 0 Then
sec = Trim$(CStr(ws.Cells(s, 1).Value)) & " / " & Trim$(CStr(ws.Cells(s, 2).Value))
End If
If modTemp.ДобавитьВоВременнуюБазу(nm, CDbl(massV), un, sec) Then
modLayout.ПодсветитьДублиУчастка h
MsgBox "Добавлено во временную базу:" & vbCrLf & modTemp.TempFilePath(), vbInformation
End If
End Sub
